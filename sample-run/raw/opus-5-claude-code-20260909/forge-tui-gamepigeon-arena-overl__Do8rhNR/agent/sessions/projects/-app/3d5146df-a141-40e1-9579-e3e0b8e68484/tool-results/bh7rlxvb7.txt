1110	  private stackedSlot(location: Location, offset: number, layout: RestaurantArenaLayout): ArenaPoint {
1111	    const step = layout.compact ? 17 : this.px(22);
1112	    const slot = layout.waypoints[location];
1113	    // Staff at the door stand *beside* the host stand rather than on top of it:
1114	    // the stand carries the hold-the-door control, and a token parked over it
1115	    // would eat the click.
1116	    if (location !== "queue") return { x: slot.x + offset * step, y: slot.y };
1117	    const lead = layout.doorStand.width / 2 + step / 2;
1118	    const room = layout.door.x + layout.door.width - step / 2;
1119	    return { x: Math.min(slot.x + lead + offset * step, room), y: slot.y };

1178	  /**
1179	   * Drag a member of staff onto another zone to reassign them — the same
1180	   * `assign_<role>_<location>` action the keyboard and the agents use, except
1181	   * you point at where you want them.
1182	   *
1183	   * Dropping is the only part that talks to the game. Everything before it is
1184	   * local: the token follows the pointer, the zone under it lights up, and any
1185	   * repaint the shift asks for in the meantime is held by the gate so the thing
1186	   * being dragged is still there to be dropped.
1187	   */
1188	  private makeDraggable(
1189	    token: Phaser.GameObjects.Container,
1190	    role: Role,
1191	    from: Location,
1192	    layout: RestaurantArenaLayout,
1193	  ): void {
1194	    // A grab box a bit taller than the figure, so the badge over their head and
1195	    // the name tag under their feet are part of the person, not dead space.
1196	    const reach = this.px(layout.compact ? 26 : 30);
1197	    token.setInteractive(
1198	      new Phaser.Geom.Rectangle(-reach / 2, -reach * 0.85, reach, reach * 1.5),
1199	      Phaser.Geom.Rectangle.Contains,
1200	    );
1201	    this.input.setDraggable(token);
1202	    // "You can pick this person up" has to be discoverable without a legend.
1203	    if (token.input) token.input.cursor = "grab";
1204	
1205	    const halo = this.add.rectangle(0, 0, 0, 0).setOrigin(0, 0).setDepth(19).setVisible(false);
1206	    halo.setStrokeStyle(2, COLORS.pass, 0.95);
1207	
1208	    token.on("dragstart", () => {
1209	      this.repaintGate.beginDrag();
1210	      this.draggingRole = role;
1211	      const walker = this.walkers.get(role);
1212	      if (walker) walker.dragging = true;
1213	      // The idle bob is a tween on this container's y; it would fight the
1214	      // pointer for the token.
1215	      this.tweens.killTweensOf(token);
1216	      token.setDepth(40);
1217	      token.setAlpha(0.92);
1218	    });
1219	
1220	    token.on("drag", (_pointer: Phaser.Input.Pointer, dragX: number, dragY: number) => {
1221	      token.setPosition(dragX, dragY);
1222	      const walker = this.walkers.get(role);
1223	      if (walker) {
1224	        walker.x = dragX;
1225	        walker.y = dragY;
1226	      }
1227	      const drop = this.zoneAt(dragX, dragY, layout);
1228	      const rect = drop && drop !== from ? this.zoneRect(drop, layout) : undefined;
1229	      if (rect) {
1230	        halo.setPosition(rect.x, rect.y).setSize(rect.width, rect.height).setVisible(true);
1231	      } else {
1232	        halo.setVisible(false);
1233	      }
1234	    });
1235	
1236	    token.on("dragend", () => {
1237	      halo.setVisible(false);
1238	      token.setAlpha(1);
1239	      token.setDepth(20);
1240	      const walker = this.walkers.get(role);
1241	      if (walker) walker.dragging = false;
1242	      this.draggingRole = undefined;
1243	      const drop = this.zoneAt(token.x, token.y, layout);
1244	      const reassigned = drop !== undefined && drop !== from;
1245	      if (reassigned) {
1246	        // Left standing where they were dropped, in the zone they were dropped
1247	        // into, so the walk that follows is the short one onto their slot
1248	        // rather than a march back out of the room they just left.
1249	        walker?.snap({ x: token.x, y: token.y }, drop);
1250	        this.onAction(`assign_${role}_${drop}`);
1251	      }
1252	      // Whatever the shift asked for while the token was in the air happens
1253	      // now — including the repaint for the assignment just dispatched.
1254	      const owed = this.repaintGate.endDrag();
1255	      if (owed || reassigned) this.paint();
1256	      // Dropped where they already were, or off the floor entirely: they walk
1257	      // back to their station rather than staying stuck under the cursor.
1258	      else this.walkHome(role, from, layout);
1259	    });
1260	  }

1301	  /**
1302	   * Walks in progress. Tokens are moved in place rather than repainted: a walk
1303	   * is sixty frames of motion, and repainting the floor for each of them would
1304	   * throw away every hotspot under the player's cursor sixty times a second.
1305	   */
1306	  update(_time: number, delta: number): void {
1307	    const layout = this.layout;
1308	    if (!layout) return;
1309	    // Walks run at the shift's own speed, so 4× service looks like 4× service.
1310	    // A stopped clock still lets a walk finish, though: a token left frozen
1311	    // halfway across the floor — after a drag during a decision, say — reads as
1312	    // a bug rather than as a pause.
1313	    const speed = this.running ? (this.state.sim?.speed ?? 1) : 1;
1314	    for (const [role, walker] of this.walkers) {
1315	      const token = this.staffTokens.get(role);
1316	      if (role === this.draggingRole) continue;
1317	      const before = walker.walking;
1318	      if (!walker.step(delta, speed)) {
1319	        // The hop that just landed leaves the token parked; give it its idle
1320	        // breathing back so it doesn't read as frozen.
1321	        if (before && !walker.walking && token) this.startIdleBob(token, walker.y, role);
1322	        continue;
1323	      }
1324	      token?.setPosition(walker.x, walker.y);
1325	      const figure = token?.getByName("figure");
1326	      if (figure && "setScale" in figure) {
1327	        (figure as Phaser.GameObjects.Container).setScale(walker.facing, 1);
1328	      }
1329	    }
1330	  }

1343	    }

1558	  /**
1559	   * A brief callout for what a decision just changed — dead centre, right
1560	   * where the decision card itself was, since that's where the player's
1561	   * eyes already are the instant they click an option. The service log
1562	   * still gets the flavour line; this is the one-line receipt so the
1563	   * consequence doesn't require scrolling to find.
1564	   *
1565	   * Every `paint()` redraws this from scratch like everything else, so the
1566	   * fade can't be a single tween that outlives one repaint — instead this
1567	   * recomputes the correct in-flight alpha from elapsed wall-clock time on
1568	   * every call (right even if a tick landed mid-fade) and hands off to one
1569	   * tween per paint for the remaining motion.
1570	   */
1571	  private drawOutcomeToast(compact: boolean, width: number, height: number): void {
1572	    const toast = this.outcomeToast;
1573	    if (!toast) return;
1574	    const elapsed = Date.now() - toast.startedAt;
1575	    if (elapsed >= this.outcomeToastMs) {
1576	      this.outcomeToast = undefined;
1577	      return;
1578	    }
1579	    const tone = toast.tone === "safe" ? 0x2f5c46 : toast.tone === "risky" ? 0x6b2f2b : 0x40342a;
1580	
1581	    const label = this.add.text(0, 0, toast.headline, {
1582	      color: "#fff3d6",
1583	      fontFamily: MONO,
1584	      fontSize: compact ? this.fs(8) : this.fs(10),
1585	      fontStyle: "bold",
1586	    });
1587	    const padX = compact ? 10 : 14;
1588	    const padY = compact ? 6 : 8;
1589	    const boxWidth = label.width + padX * 2;
1590	    const boxHeight = label.height + padY * 2;
1591	    const x = (width - boxWidth) / 2;
1592	    const y = height * 0.34 - boxHeight / 2;
1593	
1594	    const bg = this.add.rectangle(0, 0, boxWidth, boxHeight, tone).setOrigin(0, 0);
1595	    bg.setStrokeStyle(2, COLORS.pass, 0.9);
1596	    label.setPosition(padX, padY);
1597	    const toastBox = this.add.container(x, y, [bg, label]).setDepth(45);
1598	
1599	    const fadeInMs = 180;
1600	    if (elapsed < fadeInMs) {
1601	      toastBox.setAlpha(elapsed / fadeInMs);
1602	      this.tweens.add({ targets: toastBox, alpha: 1, duration: fadeInMs - elapsed });
1603	    }
1604	    const remaining = this.outcomeToastMs - elapsed;
1605	    const fadeOutMs = Math.min(400, remaining);
1606	    this.tweens.add({ targets: toastBox, alpha: 0, delay: Math.max(0, remaining - fadeOutMs), duration: fadeOutMs });
1607	  }

1825	  /**
1826	   * The manager brief. Everything above the floor used to be transport
1827	   * controls and a money figure; a first-time player had to infer tonight's
1828	   * target from a "/190" suffix and infer the next move from nothing at all.
1829	   * This band tracks the three conditions that can lose the night, names the
1830	   * one thing to do now, and keeps the last decision's receipt on screen
1831	   * after its toast has faded (#304).
1832	   *
1833	   * Kept quiet on purpose. Only a condition in trouble gets colour — a band
1834	   * of green chips reads as urgent when nothing is wrong, and then has no
1835	   * louder register left for when something is. Same reason the tag does not
1836	   * pulse for an open decision: the card is already modal and the top bar
1837	   * already flashes NEEDS A CALL, so a third alarm only adds noise.
1838	   *
1839	   * The content is computed by `arenaBrief`, which is pure and tested; this
1840	   * only lays it out. The training coach feeds the same "now" line rather
1841	   * than getting a banner of its own, so there is exactly one place on
1842	   * screen that answers "what do I do".
1843	   */
